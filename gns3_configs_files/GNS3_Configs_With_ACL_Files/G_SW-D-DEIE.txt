G_SW-D-DEIE

enable
configure terminal
hostname SW-D-DEIE
ip routing

vlan 10
name VLAN-DEIE
exit

vlan 99
name VLAN-MGMT
exit

vlan 100
name NATIVE-VLAN
exit

interface Vlan 10
description VLAN-DEIE
ip helper-address 10.10.40.2
ip address 10.10.10.1 255.255.255.0
no shutdown
exit

interface Vlan 99
description VLAN-MGMT
ip address 10.99.99.11 255.255.255.0
no shutdown
exit

interface GigabitEthernet0/1
description SW-CORE-L3-Link
no switchport
ip address 10.255.0.10 255.255.255.252
no shutdown
exit

interface GigabitEthernet0/0
description SW-A-DEIE-Trunk
switchport trunk encapsulation dot1q
switchport mode trunk
switchport trunk native vlan 100
switchport trunk allowed vlan 10,99,100
no shutdown
exit

interface Loopback0
ip address 4.4.4.4 255.255.255.255
exit

router ospf 1
router-id 4.4.4.4
network 10.255.0.8 0.0.0.3 area 0
network 10.10.10.0 0.0.0.255 area 0
network 10.99.99.0 0.0.0.255 area 0
network 4.4.4.4 0.0.0.0 area 0
exit

username admin secret admin@1234
ip domain-name uor.lk
ip ssh version 2
crypto key generate rsa
1024
service password-encryption

line vty 0 4
login local
exec-timeout 10
transport input ssh
exit

exit
write memory

---------------------------------------------------ACL

enable
configure terminal

ip access-list standard ACL_MGMT_SSH
! Permit SSH management restricted to VLAN 99
permit 10.99.99.0 0.0.0.255
! Default deny explicitly acknowledged
deny any

line vty 0 4
access-class ACL_MGMT_SSH in
exit
do write memory


enable
configure terminal

ip access-list extended ACL_DEIE_IN
! Essential Network Services
permit udp any any eq bootps
permit udp any any eq bootpc
! Engineering staff have full server farm access
permit ip 10.10.10.0 0.0.0.255 10.10.40.0 0.0.0.255
! Explicitly block access to other internal departments
deny ip 10.10.10.0 0.0.0.255 10.10.20.0 0.0.0.255
deny ip 10.10.10.0 0.0.0.255 10.10.30.0 0.0.0.255
deny ip 10.10.10.0 0.0.0.255 10.99.99.0 0.0.0.255
! Permit Internet Egress
permit ip 10.10.10.0 0.0.0.255 any
! Default deny explicitly acknowledged
deny ip any any

interface Vlan 10
ip access-group ACL_DEIE_IN in
exit
do write memory
