G_R-EDGE

enable
configure terminal
hostname R-EDGE

interface Loopback0
ip address 10.255.255.1 255.255.255.255
exit

interface GigabitEthernet0/0
description WAN-OUTSIDE-InternetCloud
ip address 200.200.200.2 255.255.255.252
no shutdown
exit

interface GigabitEthernet0/1
description R-CORE-Link
ip address 10.255.0.1 255.255.255.252
no shutdown
exit

ip route 0.0.0.0 0.0.0.0 200.200.200.1

router ospf 1
router-id 1.1.1.1
network 10.255.0.0 0.0.0.3 area 0
network 10.255.255.1 0.0.0.0 area 0
default-information originate
exit

interface GigabitEthernet0/1
ip nat inside
exit

interface GigabitEthernet0/0
ip nat outside
exit

access-list 1 permit 10.10.10.0 0.0.0.255
access-list 1 permit 10.10.20.0 0.0.0.255

ip nat inside source list 1 interface GigabitEthernet0/0 overload

username admin secret admin@1234
ip domain-name uor.lk
ip ssh version 2
crypto key generate rsa 
1024
service password-encryption

line vty 0 4
login local
exec-timeout 10
transport input ssh
exit

exit
write memory


----------------------------------------ACL--------------------

enable
configure terminal

ip access-list standard ACL_MGMT_SSH
! Permit SSH management restricted to VLAN 99
permit 10.99.99.0 0.0.0.255
! Default deny explicitly acknowledged
deny any

line vty 0 4
access-class ACL_MGMT_SSH in
exit
do write memory
