G_SW-A-DIS


enable
configure terminal
hostname SW-A-DIS

vlan 40
name VLAN-DIS
exit

vlan 99
name VLAN-MGMT
exit

vlan 100
name NATIVE-VLAN
exit

interface vlan 99
description VLAN-MGMT
ip address 10.99.99.24 255.255.255.0
no shutdown
exit

ip default-gateway 10.99.99.1

interface GigabitEthernet1/0
description SW-CORE
switchport trunk encapsulation dot1q
switchport mode trunk
switchport trunk native vlan 100
switchport trunk allowed vlan 40,99,100
no shutdown
exit

interface GigabitEthernet0/1
description DIS-PC-01
switchport mode access
switchport access vlan 40
spanning-tree portfast
spanning-tree bpduguard enable
no shutdown
exit

interface GigabitEthernet0/2
description DIS-PC-02
switchport mode access
switchport access vlan 40
spanning-tree portfast
spanning-tree bpduguard enable
no shutdown
exit

interface GigabitEthernet0/3
description DHCP/WEB
switchport mode access
switchport access vlan 40
spanning-tree portfast
spanning-tree bpduguard enable
no shutdown
exit

interface GigabitEthernet1/1
description Auto-VM
switchport mode access
switchport access vlan 40
spanning-tree portfast
spanning-tree bpduguard enable
no shutdown
exit

interface GigabitEthernet1/2
description Zabbix-VM
switchport mode access
switchport access vlan 40
spanning-tree portfast
spanning-tree bpduguard enable
no shutdown
exit


username admin secret admin@1234
ip domain-name uor.lk
ip ssh version 2
crypto key generate rsa
1024
service password-encryption

line vty 0 4
login local
exec-timeout 10
transport input ssh
exit

exit
write memory


enable
configure terminal

! Fix DHCP/WEB Port
interface GigabitEthernet0/3
no spanning-tree bpduguard enable
shutdown
no shutdown
exit

! Fix Auto-VM Port
interface GigabitEthernet1/1
no spanning-tree bpduguard enable
shutdown
no shutdown
exit

! Preemptively fix Zabbix-VM Port
interface GigabitEthernet1/2
no spanning-tree bpduguard enable
shutdown
no shutdown
exit

do write memory


---------------------------------------------------ACL

enable
configure terminal

ip access-list standard ACL_MGMT_SSH
! Permit SSH management restricted to VLAN 99
permit 10.99.99.0 0.0.0.255
! Default deny explicitly acknowledged
deny any

line vty 0 4
access-class ACL_MGMT_SSH in
exit
do write memory
