enable
configure terminal
hostname SW-CORE
ip routing

interface GigabitEthernet0/0
description R-CORE-Link
no switchport
ip address 10.255.0.6 255.255.255.252
no shutdown
exit


interface GigabitEthernet0/1
description SW-D-DEIE-Link
no switchport
ip address 10.255.0.9 255.255.255.252
no shutdown
exit


interface GigabitEthernet0/2
description SW-D-DCEE-Link
no switchport
ip address 10.255.0.13 255.255.255.252
no shutdown
exit


interface GigabitEthernet0/3
description SW-D-DMME-Link
no switchport
ip address 10.255.0.17 255.255.255.252
no shutdown
exit


interface GigabitEthernet1/0
description SW-A-DIS-Link
switchport trunk encapsulation dot1q
switchport mode trunk
switchport trunk native vlan 100
switchport trunk allowed vlan 40,99,100
no shutdown
exit

vlan 40
name VLAN_DIS
exit

vlan 99
name VLAN_MGMT
exit

vlan 100
name NATIVE-VLAN
exit

interface Vlan 40
description VLAN-DIS
ip helper-address 10.10.40.2
ip address 10.10.40.1 255.255.255.0
no shutdown
exit

interface Vlan 99
description VLAN-MGMT
ip address 10.99.99.1 255.255.255.0
no shutdown
exit

interface Loopback0
ip address 10.255.255.3 255.255.255.255
exit

router ospf 1
router-id 3.3.3.3
network 10.255.0.4 0.0.0.3 area 0
network 10.255.0.8 0.0.0.3 area 0
network 10.255.0.12 0.0.0.3 area 0
network 10.255.0.16 0.0.0.3 area 0
network 10.10.40.0 0.0.0.255 area 0
network 10.99.99.0 0.0.0.255 area 0
network 10.255.255.3 0.0.0.0 area 0
exit

username admin secret admin@1234
ip domain-name uor.lk
ip ssh version 2
crypto key generate rsa
1024
service password-encryption

line vty 0 15
login local
exec-timeout 10
transport input ssh
exit

exit
write memory

enable
configure terminal

ip access-list standard ACL_MGMT_SSH
! Permit SSH management restricted to VLAN 99
permit 10.99.99.0 0.0.0.255
! Default deny explicitly acknowledged
deny any

line vty 0 4
access-class ACL_MGMT_SSH in
exit
do write memory


enable
configure terminal

ip access-list extended ACL_DIS_IN
! Monitoring agent must reach all devices via SNMP (UDP 161)
permit udp host 10.10.40.5 any eq snmp
! Allow Server Farm to reply to permitted internal traffic
permit ip 10.10.40.0 0.0.0.255 10.10.0.0 0.255.255.255
permit ip 10.10.40.0 0.0.0.255 10.99.99.0 0.0.0.255
! Default deny explicitly acknowledged
deny ip any any

interface Vlan 40
ip access-group ACL_DIS_IN in
exit
do write memory
