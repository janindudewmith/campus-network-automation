G_SW-D-DMME

enable
configure terminal
hostname SW-D-DMME

ip routing

vlan 30
name VLAN-DMME
exit

vlan 99
name VLAN-MGMT
exit

vlan 100
name NATIVE-VLAN
exit

interface Vlan 30
description VLAN-DMME
ip helper-address 10.10.40.2
ip address 10.10.30.1 255.255.255.0
no shutdown
exit

interface Vlan 99
description VLAN-MGMT
ip address 10.99.99.13 255.255.255.0
no shutdown
exit

interface GigabitEthernet0/3
description SW-CORE-Link
no switchport
ip address 10.255.0.18 255.255.255.252
no shutdown
exit

interface GigabitEthernet0/0
description SW-A-DMME
encapsulation dot1q
switchport mode trunk
switchport trunk native vlan 100
switchport trunk allowed vlan 30,99,100
no shutdown
exit

interface Loopback0
ip address 6.6.6.6 255.255.255.255
exit

router ospf 1
router-id 6.6.6.6
network 10.255.0.16 0.0.0.3 area 0
network 10.10.30.0 0.0.0.255 area 0
network 10.99.99.0 0.0.0.255 area 0
network 6.6.6.6 0.0.0.0 area 0
exit

username admin secret admin@1234
ip domain-name uor.lk
ip ssh version 2
crypto key generate rsa
1024
service password-encryption

line vty 0 4
login local
exec-timeout 10
transport input ssh
exit

exit
write memory

enable
configure terminal
interface GigabitEthernet0/0
switchport trunk encapsulation dot1q
switchport mode trunk
exit
do write memory


---------------------------------------------------ACL

enable
configure terminal

ip access-list standard ACL_MGMT_SSH
! Permit SSH management restricted to VLAN 99
permit 10.99.99.0 0.0.0.255
! Default deny explicitly acknowledged
deny any

line vty 0 4
access-class ACL_MGMT_SSH in
exit
do write memory

enable
configure terminal

ip access-list extended ACL_DMME_IN
! Essential Network Services
permit udp any any eq bootps
permit udp any any eq bootpc
! Workshop lab has no server farm access
deny ip 10.10.30.0 0.0.0.255 10.10.40.0 0.0.0.255
! Workshop is fully isolated from admin
deny ip 10.10.30.0 0.0.0.255 10.10.20.0 0.0.0.255
! Default deny explicitly acknowledged
deny ip any any

interface Vlan 30
ip access-group ACL_DMME_IN in
exit
do write memory
