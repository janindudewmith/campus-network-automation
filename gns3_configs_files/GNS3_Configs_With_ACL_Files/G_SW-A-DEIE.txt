G_SW-A-DEIE


enable
configure terminal
hostname SW-A-DEIE

vlan 10
name VLAN-DEIE
exit

vlan 99
name VLAN-MGMT
exit

vlan 100
name NATIVE-VLAN
exit

interface vlan 99
description VLAN-MGMT
ip address 10.99.99.21 255.255.255.0
no shutdown
exit

ip default-gateway 10.99.99.1

interface GigabitEthernet0/0
description SW-D-DEIE-Trunk
switchport trunk encapsulation dot1q
switchport mode trunk
switchport trunk native vlan 100
switchport trunk allowed vlan 10,99,100
no shutdown
exit

interface GigabitEthernet0/1
description DEIE-PC-01
switchport mode access
switchport access vlan 10
spanning-tree portfast
spanning-tree bpduguard enable
no shutdown
exit

interface GigabitEthernet0/2
description DEIE-PC-02
switchport mode access
switchport access vlan 10
spanning-tree portfast
spanning-tree bpduguard enable
no shutdown
exit

username admin secret admin@1234
ip domain-name uor.lk
ip ssh version 2
crypto key generate rsa
1024
service password-encryption

line vty 0 4
login local
exec-timeout 10
transport input ssh
exit

exit
write memory


enable
configure terminal
no ip default-gateway 10.99.99.1
ip default-gateway 10.99.99.11
exit
write memory


---------------------------------------------------ACL

enable
configure terminal

ip access-list standard ACL_MGMT_SSH
! Permit SSH management restricted to VLAN 99
permit 10.99.99.0 0.0.0.255
! Default deny explicitly acknowledged
deny any

line vty 0 4
access-class ACL_MGMT_SSH in
exit
do write memory
